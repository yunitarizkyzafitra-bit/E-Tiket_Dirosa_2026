const GAS_URL = "https://script.google.com/macros/s/AKfycbwYR-okEx18_jMGBr7fNWbfhMnK8vYV7dsbDqNTm9_lVwU9R_FDq6yBGZX-l0JcKwkN_A/exec"; async function init() { const res = await fetch(GAS_URL); const data = await res.json(); let sold = data.filter(r => r[4] === 'Terjual').length; new Chart(document.getElementById('myChart'), { type: 'doughnut', data: { labels: ['Terjual', 'Sisa'], datasets: [{ data: [sold, 450-sold], backgroundColor: ['#ef4444', '#10b981'] }] } }); const grid = document.getElementById('grid'); for(let i = 1; i <= 450; i++) { let rowData = data.find(r => r[0] == i); let status = rowData ? rowData[4] : "Tersedia"; let div = document.createElement('div'); div.className = `ticket-box ${status == 'Terjual' ? 'bg-merah' : 'bg-hijau'}`; div.innerText = i; div.style.cursor = "pointer"; // Memberikan indikasi bahwa bisa diklik // Logika Klik untuk input instan div.onclick = function() { if(status === 'Terjual') { alert("Tiket nomor " + i + " sudah terjual."); } else { if(confirm("Apakah Anda ingin menandai tiket " + i + " sebagai Terjual?")) { quickSave(i); } } }; grid.appendChild(div); } async function quickSave(nomorTiket) { let p = { no: nomorTiket, tgl: new Date().toISOString().split('T')[0], // Otomatis tanggal hari ini pj: "Admin", // Default PJ target: "Umum", // Default Target status: "Terjual", ket: "Input Instan", hp: "-", alamat: "-" }; await fetch(GAS_URL, { method: 'POST', body: JSON.stringify(p) }); alert("Tiket " + nomorTiket + " berhasil terjual!"); location.reload(); // Refresh untuk update warna grid } window.onload = init;
